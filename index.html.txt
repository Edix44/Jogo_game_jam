        function playSound(type) {
            if (isMuted || !audioCtx) return;
            try {
                const now = audioCtx.currentTime;
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.connect(gain);
                gain.connect(audioCtx.destination);

                if (type === 'pickup') {
                    // Som rápido agudo ao pegar uma ferramenta
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(520, now);
                    osc.frequency.exponentialRampToValueAtTime(880, now + 0.1);
                    gain.gain.setValueAtTime(0.15, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.12);
                    osc.start(now);
                    osc.stop(now + 0.12);
                } else if (type === 'paper') {
                    // Som ao abrir a folha de cola
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(300, now);
                    osc.frequency.linearRampToValueAtTime(450, now + 0.08);
                    gain.gain.setValueAtTime(0.12, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.1);
                    osc.start(now);
                    osc.stop(now + 0.1);
                } else if (type === 'cure') {
                    // Som alegre quando cura um problema do paciente
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(523.25, now);       // C5
                    osc.frequency.setValueAtTime(659.25, now + 0.08);// E5
                    osc.frequency.setValueAtTime(783.99, now + 0.16);// G5
                    osc.frequency.setValueAtTime(1046.5, now + 0.24);// C6
                    gain.gain.setValueAtTime(0.2, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.4);
                    osc.start(now);
                    osc.stop(now + 0.4);
                } else if (type === 'error') {
                    // Buzzer grave quando usa o instrumento errado (-5s)
                    osc.type = 'sawtooth';
                    osc.frequency.setValueAtTime(160, now);
                    osc.frequency.linearRampToValueAtTime(110, now + 0.3);
                    gain.gain.setValueAtTime(0.22, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.35);
                    osc.start(now);
                    osc.stop(now + 0.35);
                } else if (type === 'nextPatient') {
                    // Fanfarra de paciente salvo!
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(440, now);
                    osc.frequency.setValueAtTime(554.37, now + 0.1);
                    osc.frequency.setValueAtTime(659.25, now + 0.2);
                    osc.frequency.setValueAtTime(880, now + 0.3);
                    gain.gain.setValueAtTime(0.25, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.55);
                    osc.start(now);
                    osc.stop(now + 0.55);
                } else if (type === 'heartbeat') {
                    // Batimento cardíaco nos últimos 15 segundos
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(90, now);
                    osc.frequency.exponentialRampToValueAtTime(45, now + 0.12);
                    gain.gain.setValueAtTime(0.2, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.14);
                    osc.start(now);
                    osc.stop(now + 0.14);
                } else if (type === 'powerup') {
                    // Som vibrante de coleta de Power-up Verde
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(440, now);
                    osc.frequency.exponentialRampToValueAtTime(880, now + 0.18);
                    gain.gain.setValueAtTime(0.18, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.2);
                    osc.start(now);
                    osc.stop(now + 0.2);
                } else if (type === 'hazard') {
                    // Som de alerta de Acidente Hospitalar
                    osc.type = 'sawtooth';
                    osc.frequency.setValueAtTime(280, now);
                    osc.frequency.linearRampToValueAtTime(130, now + 0.22);

                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.25);
                    osc.start(now);
                    osc.stop(now + 0.25);
                }
            } catch (e) {
                // Ignora caso o navegador bloqueie áudio antes de interação
            }

        // Encontra uma célula de corredor (valor 0) livre para posicionar acidentes, power-ups ou NPCs
        function getRandomCorridorPosition(existingEntities = [], minPlayerDist = 75) {
            const candidates = [];
            for (let r = 0; r < ROWS; r++) {
                for (let c = 0; c < COLS; c++) {
                    if (HOSPITAL_MAP[r][c] === 0) {
                        const px = (c + 0.5) * TILE_SIZE;
                        const py = (r + 0.5) * TILE_SIZE;
                        const distToPlayer = Math.hypot(px - player.x, py - player.y);
                        if (distToPlayer >= minPlayerDist) {
                            const tooClose = existingEntities.some(e => Math.hypot(px - e.x, py - e.y) < 36);
                            if (!tooClose) {
                                candidates.push({ x: px, y: py });
                            }
                        }
                    }
                }
            }
            if (candidates.length === 0) {
                return { x: 3.5 * TILE_SIZE, y: 11.5 * TILE_SIZE };
            }
            return candidates[Math.floor(Math.random() * candidates.length)];
        }

            timeLeft = 60.0; // Sempre 60 segundos de vida!
            player.x = 12.5 * TILE_SIZE;
            player.y = 3.2 * TILE_SIZE;
            player.heldTool = null;
            hasInspectedCurrentPatient = false;

            // 1. Sorteia 'numProblems' condições médicas diferentes dentre as 20
            const shuffledConditions = [...MEDICAL_DATABASE].sort(() => Math.random() - 0.5);
            activeProblems = shuffledConditions.slice(0, numProblems);

            // 2. Coleta todos os pisos livres (valor 0 na matriz) para espalhar os 20 aparelhos
            const validFloorTiles = [];
            for (let r = 0; r < ROWS; r++) {
                for (let c = 0; c < COLS; c++) {
                    if (HOSPITAL_MAP[r][c] === 0) {
                        validFloorTiles.push({ r, c });
                    }
                }
            }

            // Embaralha os pisos livres e posiciona todas as 20 ferramentas no mapa
            validFloorTiles.sort(() => Math.random() - 0.5);
            floorTools = MEDICAL_DATABASE.map((toolData, idx) => {
                const tile = validFloorTiles[idx];
                return {
                    x: (tile.c + 0.5) * TILE_SIZE,
                    y: (tile.r + 0.5) * TILE_SIZE,
                    toolData: toolData
                };
            });

            // 3. Configura Médicos/Enfermeiros vagando (Contorno Amarelo)
            const npcCount = getNPCCountForPatient(currentPatientIndex);
            spawnRoamingNPCs(npcCount);

            // 4. Configura Acidentes Hospitalares (Contorno Vermelho) a partir do Paciente 2
            const accidentCount = getAccidentCountForPatient(currentPatientIndex);
            hospitalAccidents = [];
            for (let i = 0; i < accidentCount; i++) {
                const acc = createRandomAccident(hospitalAccidents);
                // Desfasa levemente os timers iniciais para não trocarem todos no mesmo milissegundo
                acc.timer = 10.0;
                hospitalAccidents.push(acc);
            }

            // 5. Configura os 3 Power-Ups simultâneos (Contorno Verde) desde o Paciente 1
            activePowerUps = [];
            for (let i = 0; i < 3; i++) {
                const pu = createRandomPowerUp(activePowerUps);
                pu.timer = 10.0;
                activePowerUps.push(pu);
            }

            updateHUD();
        }

        /**
         * ============================================================================
         * 6. SISTEMA DE INTERAÇÃO (TECLA 'E' OU 'ESPAÇO')
         * ============================================================================
         */
        function getClosestInteractable() {
            const distToBed = Math.hypot(player.x - bedPosition.x, player.y - bedPosition.y);
            const distToManual = Math.hypot(player.x - manualPosition.x, player.y - manualPosition.y);

            // 1. Prioridade se estiver colado no Painel da Folha de Consulta
            if (distToManual < 54) {
                return { type: 'MANUAL', dist: distToManual };
            }

            // 2. Se estiver perto da Maca do Paciente
            if (distToBed < 62) {
                return { type: 'BED', dist: distToBed };
            }

            // 3. Verifica se há alguma ferramenta médica no chão por perto
            let closestToolIndex = -1;
            let minToolDist = 42;
            for (let i = 0; i < floorTools.length; i++) {
                const d = Math.hypot(player.x - floorTools[i].x, player.y - floorTools[i].y);
                if (d < minToolDist) {
                    minToolDist = d;
                    closestToolIndex = i;
                }
            }

            if (closestToolIndex !== -1) {
                return { type: 'TOOL', index: closestToolIndex, item: floorTools[closestToolIndex] };
            }

            return null;
        }

        function handleInteractAction() {
            if (gameState !== 'PLAYING') return;

            // Se a Folha de Cola estiver aberta, apertar E ou Espaço fecha a folha
            const manualModal = document.getElementById('manualModal');
            if (!manualModal.classList.contains('hidden')) {
                closeManual();
                return;
            }

            const target = getClosestInteractable();
            if (!target) return;

            // INTERAÇÃO A: ABRIR A FOLHA DE COLA
            if (target.type === 'MANUAL') {
                openManual();
                return;
            }

            // INTERAÇÃO B: PEGAR OU TROCAR FERRAMENTA NO CHÃO (MÁX 1 NA MÃO)
            if (target.type === 'TOOL') {
                const floorItem = floorTools[target.index];
                const pickedTool = floorItem.toolData;

                if (player.heldTool === null) {
                    // Mão vazia: pega o item do chão
                    player.heldTool = pickedTool;
                    floorTools.splice(target.index, 1);
                } else {
                    // Já tem 1 item na mão: troca pelo do chão
                    const previousTool = player.heldTool;
                    player.heldTool = pickedTool;
                    floorTools[target.index].toolData = previousTool;
                }

                playSound('pickup');
                addFloatingText(player.x, player.y - 20, `Pegou: ${pickedTool.toolIcon} ${pickedTool.toolName}`, '#38bdf8');
                updateHUD();
                return;
            }

            // INTERAÇÃO C: USAR FERRAMENTA NO PACIENTE OU EXAMINAR
            if (target.type === 'BED') {
                hasInspectedCurrentPatient = true;

                if (!player.heldTool) {
                    playSound('paper');
                    addFloatingText(bedPosition.x, bedPosition.y - 20, "🔍 Paciente Examinado! Pegue um item!", "#facc15");
                    updateHUD();
                    return;
                }

                // Verifica se o instrumento na mão resolve algum dos problemas ativos do paciente
                const matchIndex = activeProblems.findIndex(prob => prob.id === player.heldTool.id);

                if (matchIndex !== -1) {
                    // ACERTOU A FERRAMENTA!
                    const curedProblem = activeProblems[matchIndex];
                    activeProblems.splice(matchIndex, 1);
                    player.heldTool = null; // Ferramenta foi consumida no tratamento
                    totalSymptomsCured++;
                    score += 150 + Math.floor(timeLeft * 2);

                    playSound('cure');
                    addFloatingText(bedPosition.x, bedPosition.y - 25, `✅ CURADO: ${curedProblem.symptom}!`, '#4ade80');

                    // Verifica se todos os problemas deste paciente foram resolvidos
                    if (activeProblems.length === 0) {
                        patientsSavedCount++;
                        score += 500;
                        playSound('nextPatient');

                        // Se salvou o 8º paciente, VENCEU O JOGO!
                        if (currentPatientIndex >= MAX_PATIENTS) {
                            triggerEndGame(true);
                            return;
                        }

                        // Avança para o próximo paciente (máximo de 7 problemas a partir do 5º)
                        currentPatientIndex++;
                        problemsPerPatient = getProblemsForPatient(currentPatientIndex);
                        addFloatingText(
                            player.x,
                            player.y - 35,
                            `🎉 PACIENTE SALVO! PRÓXIMO (#${currentPatientIndex}/8): ${problemsPerPatient} PROBLEMAS!`,
                            '#facc15'
                        );
                        setupPatientRound(problemsPerPatient);
                    } else {
                        updateHUD();
                    }
                } else {
                    // ERROU A FERRAMENTA! Penalidade de -5 segundos!
                    timeLeft = Math.max(0, timeLeft - 5.0);
                    playSound('error');
                    addFloatingText(bedPosition.x, bedPosition.y - 25, `❌ ITEM ERRADO! -5.0s`, '#f43f5e');

                    // Efeito de tremor na tela
                    const container = document.getElementById('gameContainer');
                    container.classList.remove('screen-shake');
                    void container.offsetWidth;
                    container.classList.add('screen-shake');

                    updateHUD();
                }
            }
        }

        /**
         * ============================================================================
         * 9. FÍSICA, MOVIMENTAÇÃO, NPCS, ACIDENTES E POWER-UPS
         * ============================================================================
         */
        function isWallAt(px, py, radius, checkVolatileBlockade = false) {
            // Verifica os 4 cantos da caixa delimitadora
            const points = [
                { x: px - radius, y: py - radius },
                { x: px + radius, y: py - radius },
                { x: px - radius, y: py + radius },
                { x: px + radius, y: py + radius }
            ];

            for (const pt of points) {
                const col = Math.floor(pt.x / TILE_SIZE);
                const row = Math.floor(pt.y / TILE_SIZE);
                if (row < 0 || row >= ROWS || col < 0 || col >= COLS) return true;
                const tile = HOSPITAL_MAP[row][col];
                // Parede (1), Maca (3) e Painel (4) são sólidos
                if (tile === 1 || tile === 3 || tile === 4) {
                    return true;
                }
            }

            // Se checkVolatileBlockade estiver ativo e o jogador NÃO estiver com Calmante,
            // Compostos Voláteis explodidos (últimos 8s dos 10s de vida) trancam a passagem!
            if (checkVolatileBlockade && playerEffects.calmante <= 0) {
                for (const acc of hospitalAccidents) {
                    if (acc.type === 'VOLATILE' && acc.timer <= 8.0) {
                        const dist = Math.hypot(px - acc.x, py - acc.y);
                        if (dist < radius + acc.radius) {
                            return true;
                        }
                    }
                }
            }

            return false;
        }

        function updateGame(dt) {
            if (gameState !== 'PLAYING') return;

            // 1. Atualiza o Cronômetro de 60 Segundos
            timeLeft -= dt;
            if (timeLeft <= 0) {
                timeLeft = 0;
                triggerEndGame(false);
                return;
            }

            // Formatação e alerta visual quando faltam menos de 15 segundos
            const timerDisplay = document.getElementById('timerDisplay');
            const timerBox = document.getElementById('timerBox');
            timerDisplay.textContent = `${timeLeft.toFixed(1)}s`;

            if (timeLeft <= 15.0) {
                timerBox.classList.add('timer-urgent');
                timerDisplay.className = "font-mono-timer text-2xl sm:text-3xl font-extrabold text-rose-400 leading-none";
                heartbeatTimer -= dt;
                if (heartbeatTimer <= 0) {
                    playSound('heartbeat');
                    heartbeatTimer = timeLeft < 7 ? 0.45 : 0.8;
                }
            } else {
                timerBox.classList.remove('timer-urgent');
                timerDisplay.className = "font-mono-timer text-2xl sm:text-3xl font-extrabold text-emerald-400 leading-none";
            }

            // 2. Atualiza os Timers dos Efeitos do Jogador (Buffs e Debuffs)
            for (const key of Object.keys(playerEffects)) {
                if (playerEffects[key] > 0) {
                    playerEffects[key] = Math.max(0, playerEffects[key] - dt);
                }
            }
            updateEffectsHUD();

            // 3. Movimentação do Médico (apenas se a folha de consulta não estiver aberta)
            const isManualOpen = !document.getElementById('manualModal').classList.contains('hidden');
            if (!isManualOpen) {
                let dx = 0;
                let dy = 0;
                if (keys.up) dy -= 1;
                if (keys.down) dy += 1;
                if (keys.left) dx -= 1;
                if (keys.right) dx += 1;

                // Se estiver sob efeito da Nuvem Tóxica (e sem Calmante), INVERTE os controles!
                if (playerEffects.confused > 0 && playerEffects.calmante <= 0) {
                    dx = -dx;
                    dy = -dy;
                }

                // Normaliza velocidade na diagonal
                if (dx !== 0 && dy !== 0) {
                    const inv = 1 / Math.sqrt(2);
                    dx *= inv;
                    dy *= inv;
                }

                // Calcula velocidade efetiva (Café dobra, Vírus reduz pela metade)
                let effectiveSpeed = player.speed;
                if (playerEffects.coffee > 0) effectiveSpeed *= 2.0;
                if (playerEffects.slow > 0 && playerEffects.calmante <= 0) effectiveSpeed *= 0.5;

                if (dx !== 0 || dy !== 0) {
                    if (Math.abs(dx) > Math.abs(dy)) {
                        player.facing = dx > 0 ? 'right' : 'left';
                    } else {
                        player.facing = dy > 0 ? 'down' : 'up';
                    }

                    const nextX = player.x + dx * effectiveSpeed * dt;
                    const nextY = player.y + dy * effectiveSpeed * dt;

                    // Movimento separado nos eixos X e Y (verifica paredes + bloqueio de Composto Volátil)
                    if (!isWallAt(nextX, player.y, player.radius, true)) {
                        player.x = nextX;
                    }
                    if (!isWallAt(player.x, nextY, player.radius, true)) {
                        player.y = nextY;
                    }
                }
            }

            // 4. Atualiza os Médicos/Enfermeiros (Contorno Amarelo) e Colisão/Empurrão
            updateRoamingNPCs(dt);

            // 5. Atualiza os Acidentes Hospitalares (Contorno Vermelho, ciclo de 10s)
            updateHospitalAccidents(dt);

            // 6. Atualiza os 3 Power-Ups (Contorno Verde, ciclo de 10s ou coleta imediata)
            updatePowerUps(dt);

            // 7. Verifica proximidade da maca para abrir automaticamente o Diagnóstico Visual
            const distToBed = Math.hypot(player.x - bedPosition.x, player.y - bedPosition.y);
            const inspectCard = document.getElementById('patientInspectCard');

            if (distToBed < 85) {
                if (!hasInspectedCurrentPatient) {
                    hasInspectedCurrentPatient = true;
                    updateHUD();
                }
                inspectCard.classList.remove('hidden');
                const listEl = document.getElementById('patientSymptomsList');
                listEl.innerHTML = activeProblems.map(p => `
                    <li class="flex items-center gap-2 bg-slate-800/90 px-2.5 py-1.5 rounded-lg border-l-4" style="border-color:${p.color}">
                        <span class="text-base">${p.badge}</span>
                        <div>
                            <div class="text-white font-bold leading-tight">${p.symptom}</div>
                            <div class="text-[10px] text-slate-400">${p.desc}</div>
                        </div>
                    </li>
                `).join('');
            } else {
                inspectCard.classList.add('hidden');
            }

            // 8. Atualiza o Prompt de Interação na parte inferior
            const target = getClosestInteractable();
            const promptEl = document.getElementById('interactionPrompt');
            const promptText = document.getElementById('interactionText');

            if (target && !isManualOpen) {
                promptEl.classList.remove('hidden');
                if (target.type === 'MANUAL') {
                    promptText.textContent = "Abrir Folha de Cola 📋 (Ver quais itens curam o paciente)";
                } else if (target.type === 'BED') {
                    if (player.heldTool) {
                        promptText.textContent = `Usar ${player.heldTool.toolIcon} ${player.heldTool.toolName} no Paciente`;
                    } else {
                        promptText.textContent = "Examinar Paciente (Traga uma ferramenta para curar!)";
                    }
                } else if (target.type === 'TOOL') {
                    const t = target.item.toolData;
                    if (player.heldTool) {
                        promptText.textContent = `Trocar por ${t.toolIcon} ${t.toolName}`;
                    } else {
                        promptText.textContent = `Pegar ${t.toolIcon} ${t.toolName}`;
                    }
                }
            } else {
                promptEl.classList.add('hidden');
            }

            // 9. Atualiza textos flutuantes
            for (let i = floatingTexts.length - 1; i >= 0; i--) {
                floatingTexts[i].y -= 28 * dt;
                floatingTexts[i].life -= dt;
                floatingTexts[i].alpha = Math.max(0, floatingTexts[i].life / 1.4);
                if (floatingTexts[i].life <= 0) {
                    floatingTexts.splice(i, 1);
                }
            }
        }

        // Movimentação e Empurrão dos Médicos/Enfermeiros (Contorno Amarelo)
        function updateRoamingNPCs(dt) {
            const directions = [
                { vx: 1, vy: 0 },
                { vx: -1, vy: 0 },
                { vx: 0, vy: 1 },
                { vx: 0, vy: -1 }
            ];

            for (const npc of roamingNPCs) {
                npc.changeDirTimer -= dt;

                const nextX = npc.x + npc.vx * npc.speed * dt;
                const nextY = npc.y + npc.vy * npc.speed * dt;

                // Evita que os NPCs entrem na área da maca (tile !== 0) ou batam nas paredes
                const col = Math.floor(nextX / TILE_SIZE);
                const row = Math.floor(nextY / TILE_SIZE);
                const tileAtNext = (row >= 0 && row < ROWS && col >= 0 && col < COLS) ? HOSPITAL_MAP[row][col] : 1;

                if (isWallAt(nextX, nextY, npc.radius, false) || tileAtNext !== 0 || npc.changeDirTimer <= 0) {
                    // Escolhe uma nova direção livre no corredor
                    const validDirs = directions.filter(d => {
                        const testX = npc.x + d.vx * TILE_SIZE * 0.6;
                        const testY = npc.y + d.vy * TILE_SIZE * 0.6;
                        const tc = Math.floor(testX / TILE_SIZE);
                        const tr = Math.floor(testY / TILE_SIZE);
                        return tr >= 0 && tr < ROWS && tc >= 0 && tc < COLS &&
                               HOSPITAL_MAP[tr][tc] === 0 &&
                               !isWallAt(testX, testY, npc.radius, false);
                    });

                    if (validDirs.length > 0) {
                        const chosen = validDirs[Math.floor(Math.random() * validDirs.length)];
                        npc.vx = chosen.vx;
                        npc.vy = chosen.vy;
                    } else {
                        npc.vx = -npc.vx;
                        npc.vy = -npc.vy;
                    }
                    npc.changeDirTimer = 1.4 + Math.random() * 2.2;
                } else {
                    npc.x = nextX;
                    npc.y = nextY;
                }

                // Colisão física com o Jogador (tapa o corredor e empurra se não tiver Calmante)
                if (playerEffects.calmante <= 0) {
                    const dx = player.x - npc.x;
                    const dy = player.y - npc.y;
                    const dist = Math.hypot(dx, dy);
                    const minDist = player.radius + npc.radius + 2;

                    if (dist < minDist && dist > 0.001) {
                        const overlap = (minDist - dist) + 1.5;
                        const nx = dx / dist;
                        const ny = dy / dist;

                        // Empurra o jogador na direção do impacto + direção de caminhada do NPC
                        const pushX = player.x + (nx * overlap) + (npc.vx * npc.speed * dt * 0.7);
                        const pushY = player.y + (ny * overlap) + (npc.vy * npc.speed * dt * 0.7);

                        if (!isWallAt(pushX, player.y, player.radius, true)) {
                            player.x = pushX;
                        }
                        if (!isWallAt(player.x, pushY, player.radius, true)) {
                            player.y = pushY;
                        }
                    }
                }
            }
        }

        // Ciclo de 10 segundos e colisões dos Acidentes Hospitalares (Contorno Vermelho)
        function updateHospitalAccidents(dt) {
            for (let i = 0; i < hospitalAccidents.length; i++) {
                const acc = hospitalAccidents[i];
                const prevTimer = acc.timer;
                acc.timer -= dt;

                // Se completou 10 segundos no mapa, some e outro acidente aparece em outro lugar!
                if (acc.timer <= 0) {
                    hospitalAccidents[i] = createRandomAccident(hospitalAccidents);
                    continue;
                }

                // Se for um Composto Volátil que acabou de explodir aos 8.0s restantes e o jogador estava em cima, empurra-o para fora
                if (acc.type === 'VOLATILE' && prevTimer > 8.0 && acc.timer <= 8.0) {
                    playSound('hazard');
                    addFloatingText(acc.x, acc.y - 18, "💥 EXPLODIU! Passagem Trancada (8s)!", "#ef4444");
                }

                // Verifica contato do jogador com o acidente (se não estiver intangível pelo Calmante)
                if (playerEffects.calmante <= 0) {
                    const dist = Math.hypot(player.x - acc.x, player.y - acc.y);
                    const hitDist = player.radius + acc.radius;

                    if (dist < hitDist) {
                        if (acc.type === 'TOXIC') {
                            if (playerEffects.confused <= 4.3) {
                                playSound('hazard');
                                addFloatingText(player.x, player.y - 24, "☣️ NUVEM TÓXICA! Controles Invertidos (5s)!", "#f43f5e");
                            }
                            playerEffects.confused = 5.0;
                        } else if (acc.type === 'VIRUS') {
                            if (playerEffects.slow <= 4.3) {
                                playSound('hazard');
                                addFloatingText(player.x, player.y - 24, "🦠 VÍRUS EXPOSTO! Lentidão (5s)!", "#f43f5e");
                            }
                            playerEffects.slow = 5.0;
                        } else if (acc.type === 'VOLATILE' && acc.timer <= 8.0 && dist > 0.001) {
                            // Empurra o jogador para fora caso encoste na barricada da explosão
                            const nx = (player.x - acc.x) / dist;
                            const ny = (player.y - acc.y) / dist;
                            const pushX = player.x + nx * (hitDist - dist + 2);
                            const pushY = player.y + ny * (hitDist - dist + 2);
                            if (!isWallAt(pushX, player.y, player.radius, false)) player.x = pushX;
                            if (!isWallAt(player.x, pushY, player.radius, false)) player.y = pushY;
                        }
                    }
                }
            }
        }

        // Ciclo de 10 segundos e coleta em cadeia dos 3 Power-Ups (Contorno Verde)
        function updatePowerUps(dt) {
            for (let i = 0; i < activePowerUps.length; i++) {
                const pu = activePowerUps[i];
                pu.timer -= dt;

                // Se passaram 10 segundos sem coleta, some e aparece outro em outro ponto do mapa
                if (pu.timer <= 0) {
                    activePowerUps[i] = createRandomPowerUp(activePowerUps);
                    continue;
                }

                // Coleta imediata ao passar por cima
                const dist = Math.hypot(player.x - pu.x, player.y - pu.y);
                if (dist < player.radius + pu.radius) {
                    playSound('powerup');

                    if (pu.type === 'COFFEE') {
                        playerEffects.coffee = 5.0;
                        addFloatingText(player.x, player.y - 25, "☕ CAFÉ! Velocidade 2x (5s)!", "#4ade80");
                    } else if (pu.type === 'CALMANTE') {
                        playerEffects.calmante = 8.0;
                        playerEffects.confused = 0;
                        playerEffects.slow = 0;
                        addFloatingText(player.x, player.y - 25, "💊 CALMANTE! Intangível a Obstáculos (8s)!", "#4ade80");
                    } else if (pu.type === 'FOCUS') {
                        playerEffects.focus = 10.0;
                        addFloatingText(player.x, player.y - 25, "🧠 FOCO ATIVO! Itens de Cura Destacados (10s)!", "#4ade80");
                    }

                    // Assim que coleta, outro power-up aparece NA MESMA HORA em outro local aleatório!
                    activePowerUps[i] = createRandomPowerUp(activePowerUps);
                    updateHUD();
                }
            }
        }

        /**
         * ============================================================================
         * 10. RENDERIZAÇÃO GRÁFICA NO CANVAS (TOP-DOWN)
         * ============================================================================
         */
        function drawGame() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            const timeSec = performance.now() / 1000;

            // 1. Desenha os Pisos e Paredes do Hospital
            for (let r = 0; r < ROWS; r++) {
                for (let c = 0; c < COLS; c++) {
                    const tile = HOSPITAL_MAP[r][c];
                    const x = c * TILE_SIZE;
                    const y = r * TILE_SIZE;

                    if (tile === 1) {
                        // Parede Hospitalar com relevo
                        ctx.fillStyle = '#1e293b';
                        ctx.fillRect(x, y, TILE_SIZE, TILE_SIZE);
                        ctx.fillStyle = '#334155';
                        ctx.fillRect(x + 2, y + 2, TILE_SIZE - 4, TILE_SIZE - 8);
                        ctx.strokeStyle = '#0f172a';
                        ctx.lineWidth = 1;
                        ctx.strokeRect(x, y, TILE_SIZE, TILE_SIZE);
                    } else {
                        // Azulejo de piso hospitalar
                        const isChecker = (r + c) % 2 === 0;
                        if (tile === 2 || tile === 3 || tile === 4) {
                            ctx.fillStyle = isChecker ? '#0f2942' : '#12304d';
                        } else {
                            ctx.fillStyle = isChecker ? '#0f172a' : '#131c31';
                        }
                        ctx.fillRect(x, y, TILE_SIZE, TILE_SIZE);
                    }
                }
            }

            // 2. Desenha os nomes das Salas no chão
            ctx.save();
            ctx.textAlign = 'center';
            ctx.font = '900 13px Inter, sans-serif';
            for (const label of ROOM_LABELS) {
                ctx.fillStyle = label.color;
                ctx.fillText(label.text, label.x, label.y);
            }
            ctx.restore();

            // 3. Desenha o PAINEL / FOLHA DE CONSULTA (Tile 4)
            drawManualStation(manualPosition.x, manualPosition.y, timeSec);

            // 4. Desenha a MACA E O PACIENTE COM AS CORES DOS PROBLEMAS (Tile 3)
            drawPatientOnBed(bedPosition.x, bedPosition.y, timeSec);

            // 5. Desenha os ACIDENTES HOSPITALARES (CONTORNO VERMELHO)
            drawHospitalAccidents(timeSec);

            // 6. Desenha os POWER-UPS (CONTORNO VERDE)
            drawPowerUps(timeSec);

            // 7. Desenha as 20 Ferramentas Médicas espalhadas pelo Labirinto (com destaque se Foco estiver ativo!)
            for (const item of floorTools) {
                const distToPlayer = Math.hypot(player.x - item.x, player.y - item.y);
                const isNear = distToPlayer < 65;
                const isNeededForPatient = activeProblems.some(p => p.id === item.toolData.id);
                const isHighlightedByFocus = playerEffects.focus > 0 && isNeededForPatient;

                ctx.save();

                // Se o Power-Up de Foco estiver ativo e esta ferramenta curar o paciente atual, desenha um super brilho!
                if (isHighlightedByFocus) {
                    const pulseRadius = 22 + Math.sin(timeSec * 8) * 5;
                    ctx.beginPath();
                    ctx.arc(item.x, item.y, pulseRadius, 0, Math.PI * 2);
                    ctx.fillStyle = 'rgba(250, 204, 21, 0.28)';
                    ctx.fill();
                    ctx.lineWidth = 3;
                    ctx.strokeStyle = '#facc15';
                    ctx.stroke();

                    // Tag "★ CURA!" acima do item
                    ctx.font = '900 10px Inter, sans-serif';
                    ctx.textAlign = 'center';
                    ctx.fillStyle = '#facc15';
                    ctx.fillText('★ CURA!', item.x, item.y - 25);
                }

                // Base do item
                ctx.beginPath();
                ctx.arc(item.x, item.y, isNear ? 17 : 14, 0, Math.PI * 2);
                ctx.fillStyle = isHighlightedByFocus
                    ? 'rgba(234, 179, 8, 0.35)'
                    : (isNear ? 'rgba(56, 189, 248, 0.28)' : 'rgba(30, 41, 59, 0.85)');
                ctx.fill();
                ctx.lineWidth = isHighlightedByFocus ? 3 : (isNear ? 2.5 : 1.5);
                ctx.strokeStyle = isHighlightedByFocus ? '#fde047' : (isNear ? '#38bdf8' : '#475569');
                ctx.stroke();

                // Emoji do Aparelho Médico
                ctx.font = '20px sans-serif';
                ctx.textAlign = 'center';
                ctx.textBaseline = 'middle';
                const floatOffset = Math.sin(timeSec * 3 + item.x) * 2;
                ctx.fillText(item.toolData.toolIcon, item.x, item.y + floatOffset);

                // Se o médico chegar perto (ou se estiver com Foco ativo), mostra o nome do aparelho
                if (isNear || isHighlightedByFocus) {
                    ctx.font = 'bold 11px Inter, sans-serif';
                    const textWidth = ctx.measureText(item.toolData.toolName).width + 10;
                    const labelY = isHighlightedByFocus ? item.y + 24 : item.y - 32;
                    ctx.fillStyle = 'rgba(15, 23, 42, 0.92)';
                    ctx.fillRect(item.x - textWidth / 2, labelY, textWidth, 16);
                    ctx.strokeStyle = isHighlightedByFocus ? '#facc15' : '#38bdf8';
                    ctx.lineWidth = 1;
                    ctx.strokeRect(item.x - textWidth / 2, labelY, textWidth, 16);
                    ctx.fillStyle = isHighlightedByFocus ? '#fef08a' : '#f8fafc';
                    ctx.fillText(item.toolData.toolName, item.x, labelY + 8);
                }
                ctx.restore();
            }

            // 8. Desenha os Médicos/Enfermeiros vagando pelo hospital (CONTORNO AMARELO)
            drawRoamingNPCs(timeSec);

            // 9. Desenha o Jogador (Médico Substituto)
            drawDoctorPlayer(player.x, player.y, timeSec);

            // 10. Desenha os Textos Flutuantes (+CURADO, -5s ERRO, etc.)
            for (const ft of floatingTexts) {
                ctx.save();
                ctx.globalAlpha = ft.alpha;
                ctx.font = '900 14px Inter, sans-serif';
                ctx.textAlign = 'center';
                ctx.strokeStyle = '#000000';
                ctx.lineWidth = 4;
                ctx.strokeText(ft.text, ft.x, ft.y);
                ctx.fillStyle = ft.color;
                ctx.fillText(ft.text, ft.x, ft.y);
                ctx.restore();
            }
        }

        // Desenha os Acidentes Hospitalares com CONTORNO VERMELHO (#FF3333)
        function drawHospitalAccidents(timeSec) {
            for (const acc of hospitalAccidents) {
                ctx.save();
                const isVolatileArmed = acc.type === 'VOLATILE' && acc.timer > 8.0;
                const pulse = Math.sin(timeSec * 7) * 2;

                // Fundo de perigo vermelho
                ctx.beginPath();
                ctx.arc(acc.x, acc.y, acc.radius + pulse, 0, Math.PI * 2);
                if (acc.type === 'VOLATILE' && !isVolatileArmed) {
                    // Composto volátil explodido trancando passagem (barricada sólida vermelha/laranja)
                    ctx.fillStyle = 'rgba(220, 38, 38, 0.78)';
                } else {
                    ctx.fillStyle = 'rgba(239, 68, 68, 0.28)';
                }
                ctx.fill();

                // CONTORNO VERMELHO FORTE (#FF3333)
                ctx.lineWidth = 3.5;
                ctx.strokeStyle = '#ff3333';
                ctx.stroke();

                // Arco de contagem regressiva dos 10 segundos
                const ratio = Math.max(0, acc.timer / acc.maxTimer);
                ctx.beginPath();
                ctx.arc(acc.x, acc.y, acc.radius + 4, -Math.PI / 2, -Math.PI / 2 + (Math.PI * 2 * ratio));
                ctx.strokeStyle = '#fca5a5';
                ctx.lineWidth = 2;
                ctx.stroke();

                // Ícone do Acidente
                ctx.font = '20px sans-serif';
                ctx.textAlign = 'center';
                ctx.textBaseline = 'middle';
                const displayIcon = isVolatileArmed ? '🧨' : acc.icon;
                ctx.fillText(displayIcon, acc.x, acc.y);

                // Pequeno rótulo de aviso
                ctx.font = '800 9px Inter, sans-serif';
                ctx.fillStyle = '#fecdd3';
                const label = (acc.type === 'VOLATILE' && !isVolatileArmed)
                    ? `BLOQUEADO (${acc.timer.toFixed(0)}s)`
                    : `${acc.name} (${acc.timer.toFixed(0)}s)`;
                ctx.fillText(label, acc.x, acc.y + acc.radius + 10);

                ctx.restore();
            }
        }

        // Desenha os Power-Ups com CONTORNO VERDE (#00FF66)
        function drawPowerUps(timeSec) {
            for (const pu of activePowerUps) {
                ctx.save();
                const pulse = Math.sin(timeSec * 5) * 2;

                // Círculo com CONTORNO VERDE (#00FF66)
                ctx.beginPath();
                ctx.arc(pu.x, pu.y, pu.radius + pulse, 0, Math.PI * 2);
                ctx.fillStyle = 'rgba(16, 185, 129, 0.28)';
                ctx.fill();
                ctx.lineWidth = 3.5;
                ctx.strokeStyle = '#00ff66';
                ctx.stroke();

                const ratio = Math.max(0, pu.timer / pu.maxTimer);
                ctx.beginPath();
                ctx.arc(pu.x, pu.y, pu.radius + 4, -Math.PI / 2, -Math.PI / 2 + (Math.PI * 2 * ratio));
                ctx.strokeStyle = '#a7f3d0';
                ctx.lineWidth = 2;
                ctx.stroke();

                // Ícone do Power-Up (☕, 💊 ou 🧠)
                ctx.font = '19px sans-serif';
                ctx.textAlign = 'center';
                ctx.textBaseline = 'middle';

                ctx.fillText(`${pu.name} (${pu.timer.toFixed(0)}s)`, pu.x, pu.y + pu.radius + 10);

                ctx.restore();
            }
        }

                ctx.font = '19px sans-serif';

        // Desenha a bancada com a Folha de Cola ao lado do paciente
        function drawManualStation(x, y, timeSec) {
            ctx.save();
            // Mesa / Bancada
            ctx.fillStyle = '#78350f';
            ctx.strokeStyle = '#f59e0b';
            ctx.lineWidth = 2;
            ctx.fillRect(x - 18, y - 16, 36, 32);
            ctx.strokeRect(x - 18, y - 16, 36, 32);

            // Brilho pulsante convidando o jogador a consultar
            const pulse = (Math.sin(timeSec * 4) + 1) * 0.5;
            ctx.beginPath();
            ctx.arc(x, y, 22 + pulse * 5, 0, Math.PI * 2);
            ctx.strokeStyle = `rgba(251, 191, 36, ${0.3 + pulse * 0.4})`;
            ctx.lineWidth = 2;
            ctx.stroke();

            // Ícone da Folha
            ctx.font = '22px sans-serif';
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';
            ctx.fillText('📋', x, y);

            // Rótulo "FOLHA (E)"
            ctx.font = '900 10px Inter, sans-serif';
            ctx.fillStyle = '#fef3c7';
            ctx.fillText('FOLHA [E]', x, y - 24);
            ctx.restore();
        }

        // Desenha a Maca e o Paciente mudando de cor conforme os sintomas ativos!
        function drawPatientOnBed(x, y, timeSec) {
            ctx.save();

            // Base da Maca Hospitalar
            ctx.fillStyle = '#e2e8f0';
            ctx.strokeStyle = '#38bdf8';
            ctx.lineWidth = 3;
            ctx.fillRect(x - 22, y - 28, 44, 56);
            ctx.strokeRect(x - 22, y - 28, 44, 56);

            // Travesseiro
            ctx.fillStyle = '#ffffff';
            ctx.fillRect(x - 16, y - 24, 32, 12);

            // Determina as cores do corpo do paciente com base nos problemas ativos!
            const hasGreenSkin = activeProblems.some(p => p.id === 1); // Pele Verde -> Serrote
            const hasRedLimb   = activeProblems.some(p => p.id === 2); // Membro Vermelho -> Analgésicos
            const hasBlueFace  = activeProblems.some(p => p.id === 3); // Rosto Azul -> Estetoscópio

            const faceProblem  = activeProblems.find(p => p.part === 'face');
            const limbProblem  = activeProblems.find(p => p.part === 'limb');
            const chestProblem = activeProblems.find(p => p.part === 'chest');
            const feetProblem  = activeProblems.find(p => p.part === 'feet');
            const auraProblem  = activeProblems.find(p => p.part === 'aura');

            // Aura pulsante caso tenha sintoma de aura
            if (auraProblem) {
                ctx.beginPath();
                ctx.arc(x, y, 28 + Math.sin(timeSec * 6) * 4, 0, Math.PI * 2);
                ctx.fillStyle = auraProblem.color + '44';
                ctx.fill();
            }

            // Tronco do Paciente
            ctx.fillStyle = chestProblem ? chestProblem.color : (hasGreenSkin ? '#22c55e' : '#60a5fa');
            ctx.fillRect(x - 11, y - 6, 22, 20);

            // Braços / Membros (Ficam vermelhos se tiver Membro Vermelho!)
            const limbColor = hasRedLimb ? '#ef4444' : (limbProblem ? limbProblem.color : (hasGreenSkin ? '#22c55e' : '#fdba74'));
            ctx.fillStyle = limbColor;
            ctx.fillRect(x - 18, y - 5, 6, 16); // Braço esquerdo
            ctx.fillRect(x + 12, y - 5, 6, 16); // Braço direito

            // Pernas / Pés
            const feetColor = feetProblem ? feetProblem.color : (hasRedLimb ? '#ef4444' : (hasGreenSkin ? '#22c55e' : '#94a3b8'));
            ctx.fillStyle = feetColor;
            ctx.fillRect(x - 9, y + 14, 7, 11);
            ctx.fillRect(x + 2, y + 14, 7, 11);

            // Cabeça / Rosto do Paciente (Fica Verde, Azul ou da cor do sintoma facial)
            const headColor = hasBlueFace ? '#3b82f6' : (hasGreenSkin ? '#22c55e' : (faceProblem ? faceProblem.color : '#fed7aa'));
            ctx.beginPath();
            ctx.arc(x, y - 14, 9, 0, Math.PI * 2);
            ctx.fillStyle = headColor;
            ctx.fill();
            ctx.lineWidth = 1.5;
            ctx.strokeStyle = '#0f172a';
            ctx.stroke();

            // Desenha pequenas bolinhas coloridas ao redor da maca representando todos os problemas restantes
            const total = activeProblems.length;
            for (let i = 0; i < total; i++) {
                const angle = (timeSec * 1.5) + (i * (Math.PI * 2 / total));
                const bx = x + Math.cos(angle) * 29;
                const by = y + Math.sin(angle) * 29;
                ctx.beginPath();
                ctx.arc(bx, by, 5, 0, Math.PI * 2);
                ctx.fillStyle = activeProblems[i].color;
                ctx.fill();
                ctx.strokeStyle = '#0f172a';
                ctx.lineWidth = 1.5;
                ctx.stroke();
            }

            // Rótulo acima da maca
            ctx.font = '900 10px Inter, sans-serif';
            ctx.textAlign = 'center';
            ctx.fillStyle = '#fda4af';
            ctx.fillText(`PACIENTE (${activeProblems.length} PROB.)`, x, y - 34);

            ctx.restore();
        }

        // Desenha o Médico Substituto controlado pelo jogador
        function drawDoctorPlayer(x, y, timeSec) {
            ctx.save();

            // Se estiver sob efeito do Calmante (Intangível), fica semitransparente!
            if (playerEffects.calmante > 0) {
                ctx.globalAlpha = 0.55 + Math.sin(timeSec * 12) * 0.15;
            }

            // Aura de velocidade caso o Café esteja ativo
            if (playerEffects.coffee > 0) {
                ctx.beginPath();
                ctx.arc(x, y, player.radius + 6, 0, Math.PI * 2);
                ctx.strokeStyle = '#00ff66';
                ctx.lineWidth = 2;
                ctx.stroke();
            }

            // Sombra embaixo do médico
            ctx.beginPath();
            ctx.arc(x, y + 3, player.radius, 0, Math.PI * 2);
            ctx.fillStyle = 'rgba(0, 0, 0, 0.45)';
            ctx.fill();

            // Corpo (Jaleco Branco com borda Ciano ou Verde se Intangível)
            ctx.beginPath();
            ctx.arc(x, y, player.radius, 0, Math.PI * 2);
            ctx.fillStyle = '#f8fafc';
            ctx.fill();
            ctx.lineWidth = 3;
            ctx.strokeStyle = playerEffects.calmante > 0 ? '#00ff66' : (playerEffects.confused > 0 ? '#ff3333' : '#06b6d4');
            ctx.stroke();

            // Cruz vermelha de médico no jaleco
            ctx.fillStyle = '#ef4444';
            ctx.fillRect(x - 2, y - 6, 4, 12);
            ctx.fillRect(x - 6, y - 2, 12, 4);

            // Indicador de confusão (controles invertidos) ou lentidão acima da cabeça
            if (playerEffects.confused > 0 && playerEffects.calmante <= 0) {
                ctx.font = 'bold 12px sans-serif';
                ctx.textAlign = 'center';
                ctx.fillText('💫❓', x, y - 20);
            } else if (playerEffects.slow > 0 && playerEffects.calmante <= 0) {
                ctx.font = 'bold 12px sans-serif';
                ctx.textAlign = 'center';
                ctx.fillText('🦠🐌', x, y - 20);
            }

            // Se o médico estiver segurando uma ferramenta, desenha ela acima dele!
            if (player.heldTool) {
                ctx.font = '18px sans-serif';
                ctx.textAlign = 'center';
                ctx.textBaseline = 'middle';
                ctx.fillText(player.heldTool.toolIcon, x, y - 24 + Math.sin(timeSec * 6) * 2);
            }

            ctx.restore();
        }

        /**
         * ============================================================================
         * 11. EVENTOS DE TECLADO E LOOP PRINCIPAL (60 FPS)
         * ============================================================================
         */
        window.addEventListener('keydown', (e) => {
            // Se estiver digitando na busca da folha de cola, só captura ESC
            if (document.activeElement === document.getElementById('manualSearch')) {
                if (e.key === 'Escape') closeManual();
                return;
            }

            const code = e.code;
            if (code === 'KeyW' || code === 'ArrowUp')    { keys.up = true; e.preventDefault(); }
            if (code === 'KeyS' || code === 'ArrowDown')  { keys.down = true; e.preventDefault(); }
            if (code === 'KeyA' || code === 'ArrowLeft')  { keys.left = true; e.preventDefault(); }
            if (code === 'KeyD' || code === 'ArrowRight') { keys.right = true; e.preventDefault(); }

            if (code === 'KeyE' || code === 'Space') {
                e.preventDefault();
                handleInteractAction();
            }
            if (code === 'Escape') {
                closeManual();
            }
        });

        window.addEventListener('keyup', (e) => {
            const code = e.code;
            if (code === 'KeyW' || code === 'ArrowUp')    keys.up = false;
            if (code === 'KeyS' || code === 'ArrowDown')  keys.down = false;
            if (code === 'KeyA' || code === 'ArrowLeft')  keys.left = false;
            if (code === 'KeyD' || code === 'ArrowRight') keys.right = false;
        });

        function setTouchKey(dir, state) {
            keys[dir] = state;
        }

        // Loop Principal de Animação do Jogo
        function gameLoop(now) {
            const dt = Math.min(0.1, (now - lastFrameTime) / 1000);
            lastFrameTime = now;

            updateGame(dt);
            drawGame();

            requestAnimationFrame(gameLoop);
        }

        window.onload = function () {
            setupPatientRound(3);
            requestAnimationFrame(gameLoop);
        };
